import java.util.*;

// Class 1: Lista de Adjacencia
class GrafoLista {
    private final Map<Integer, List<Integer>> adjacencias = new HashMap<>();
    private final boolean eDirecionado;

    public GrafoLista(boolean eDirecionado) {
        this.eDirecionado = eDirecionado;
    }

    public void adicionarVertice(int vertice) {
        adjacencias.putIfAbsent(vertice, new ArrayList<>());
    }

    public void adicionarAresta(int origem, int destino) {
        adicionarVertice(origem);
        adicionarVertice(destino);
        adjacencias.get(origem).add(destino);
        if (!eDirecionado) {
            adjacencias.get(destino).add(origem);
        }
    }

    public void imprimirGrafo() {
        String tipo = eDirecionado ? "Direcionado" : "Não Direcionado";
        System.out.println("=== Lista de Adjacência (" + tipo + ") ===");
        for (var entry : adjacencias.entrySet()) {
            System.out.println("Vértice " + entry.getKey() + " -> " + entry.getValue());
        }
        System.out.println();
    }
}

// Class 2: Matriz de Adjacencia
class GrafoMatriz {
    private final boolean[][] matriz;
    private final int numVertices;
    private final boolean eDirecionado;

    public GrafoMatriz(int numVertices, boolean eDirecionado) {
        this.numVertices = numVertices;
        this.eDirecionado = eDirecionado;
        this.matriz = new boolean[numVertices][numVertices];
    }

    public void adicionarAresta(int origem, int destino) {
        if (origem >= 0 && origem < numVertices && destino >= 0 && destino < numVertices) {
            matriz[origem][destino] = true;
            if (!eDirecionado) {
                matriz[destino][origem] = true;
            }
        }
    }

    public void imprimirGrafo() {
        String tipo = eDirecionado ? "Direcionado" : "Não Direcionado";
        System.out.println("=== Matriz de Adjacência (" + tipo + ") ===");
        System.out.print("   ");
        for (int i = 0; i < numVertices; i++) {
            System.out.print(i + " ");
        }
        System.out.println();

        for (int i = 0; i < numVertices; i++) {
            System.out.print(i + ": ");
            for (int j = 0; j < numVertices; j++) {
                System.out.print((matriz[i][j] ? 1 : 0) + " ");
            }
            System.out.println();
        }
        System.out.println();
    }
}

// Classe Principal de Execucao (exigida pelo GDB Online)
public class Main {
    public static void main(String[] args) {
        // Teste Lista de Adjacencia
        GrafoLista listaNaoDir = new GrafoLista(false);
        listaNaoDir.adicionarAresta(0, 1);
        listaNaoDir.adicionarAresta(0, 2);
        listaNaoDir.imprimirGrafo();

        GrafoLista listaDir = new GrafoLista(true);
        listaDir.adicionarAresta(0, 1);
        listaDir.adicionarAresta(0, 2);
        listaDir.imprimirGrafo();

        // Teste Matriz de Adjacencia
        GrafoMatriz matrizNaoDir = new GrafoMatriz(3, false);
        matrizNaoDir.adicionarAresta(0, 1);
        matrizNaoDir.adicionarAresta(0, 2);
        matrizNaoDir.imprimirGrafo();

        GrafoMatriz matrizDir = new GrafoMatriz(3, true);
        matrizDir.adicionarAresta(0, 1);
        matrizDir.adicionarAresta(0, 2);
        matrizDir.imprimirGrafo();
    }
}
